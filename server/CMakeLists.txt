cmake_minimum_required(VERSION 3.22)
project(libremusic_server LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(LIBREMUSIC_BUILD_HTTP "Build the userver HTTP service" ON)
option(USERVER_FEATURE_TESTSUITE "Enable userver's optional Python testsuite" OFF)
find_package(OpenSSL REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(CURL REQUIRED)
find_package(nlohmann_json 3.10 REQUIRED)
find_package(Threads REQUIRED)
add_library(music_core src/database.cpp src/security.cpp src/process.cpp
  src/service.cpp src/jobs.cpp)
target_include_directories(music_core PUBLIC src)
target_link_libraries(music_core PUBLIC SQLite::SQLite3 OpenSSL::Crypto
  CURL::libcurl nlohmann_json::nlohmann_json Threads::Threads)
target_compile_options(music_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(music-admin src/admin.cpp)
target_link_libraries(music-admin PRIVATE music_core)
if(LIBREMUSIC_BUILD_HTTP)
  find_package(userver 3.2 REQUIRED COMPONENTS core)
  userver_setup_environment()
  add_executable(music-server src/main.cpp)
  target_link_libraries(music-server PRIVATE music_core userver::core)
endif()
enable_testing()
add_executable(music-tests tests/core_test.cpp)
target_link_libraries(music-tests PRIVATE music_core)
add_test(NAME music-core COMMAND music-tests)
